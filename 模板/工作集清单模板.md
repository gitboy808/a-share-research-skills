# 投研工作集清单模板

实际产物是运行记录旁的逐阶段 JSON sidecar，不是 Markdown 报告。文件名使用
`RUN-YYYYMMDD-NNN-WORKFLOW_SLUG-STAGE_SLUG-工作集清单.json`；同一运行的不同阶段不得覆盖。

```json
{
  "schema_version": "a-share-workspace-v3",
  "artifact_type": "workset_manifest",
  "id": "RUN-YYYYMMDD-NNN-WORKSET-WORKFLOW_UPPER-STAGE_UPPER",
  "status": "partial",
  "run_id": "RUN-YYYYMMDD-NNN",
  "workflow": "WORKFLOW_SLUG",
  "stage": "STAGE_SLUG",
  "created_at": "YYYY-MM-DDTHH:mm:ss+08:00",
  "information_cutoff": "YYYY-MM-DDTHH:mm:ss+08:00",
  "task_contract": {
    "contract_id": "CONTRACT-ID",
    "version": "CONTRACT-VERSION",
    "registry_path": "CONTRACT-REGISTRY-PATH",
    "sha256": "CONTRACT-SHA256"
  },
  "contract_instantiation": {
    "schema_version": "a-share-workspace-v3",
    "workflow": "WORKFLOW_SLUG",
    "stage": "STAGE_SLUG",
    "information_cutoff": "YYYY-MM-DDTHH:mm:ss+08:00",
    "calibration_window_start": null,
    "objects": [],
    "handoff": {},
    "strategy_version": null,
    "task_conditions": [],
    "sha256": "CONTRACT-INSTANTIATION-SHA256"
  },
  "instantiated_requirements": [],
  "instantiated_requirements_sha256": "INSTANTIATED-REQUIREMENTS-SHA256",
  "strategy_version": ["STRATEGY-ID@VERSION"],
  "stable_references": [],
  "audit_references": [],
  "audited_exclusions": [],
  "relations": [],
  "audit_relations": [],
  "relation_checks": {
    "total": 0,
    "resolved": 0,
    "blocking_gaps": 0
  },
  "verification": {
    "status": "not_run",
    "required_unit_ids": [],
    "verified_unit_ids": [],
    "missing_references": []
  },
  "projection": {
    "schema_version": "a-share-context-projection-v1",
    "content_hash": "PROJECTION-SHA256",
    "projection_degraded": false
  },
  "coverage": {
    "required_total": 0,
    "required_covered": 0,
    "required_missing": 0,
    "coverage_ratio": 0.0,
    "blocking": true,
    "blocking_gap_count": 1,
    "requirements": [],
    "semantic_candidates_do_not_count": true
  },
  "gaps": [
    {
      "requirement_id": "TASK-CONTRACT-NOT-INSTANTIATED",
      "reason": "任务契约尚未实例化；不得把零要求解释为空覆盖成功",
      "required": true,
      "allow_unknown": false,
      "blocking": true
    }
  ],
  "semantic_adapter": {
    "status": "disabled"
  },
  "budget": {},
  "quality": {
    "assembled_units": 0,
    "source_payload_bytes_in_workset": 0,
    "hydrate_units": 0,
    "projection_degraded": false,
    "context_proxy": {
      "stable_reference_characters": 0,
      "selected_source_characters": 0,
      "indexed_source_characters": 0,
      "raw_tool_payload_characters_entered": 0,
      "token_replay_available": false
    }
  }
}
```

`task_contract` 固定注册表相对路径和文件哈希；`contract_instantiation` 固定对象、handoff 与追加条件输入及其哈希；`instantiated_requirements` 必须保存由这些输入精确重算出的完整选择器及其规范 JSON 哈希，不能只保存 ID/版本。
`stable_references` 只保存可 hydrate 的稳定引用、权威产物类型、文档定位、编译后资格策略和选择原因；`audit_references` 只定位被排除的历史原子并固定 `hydrate_eligible: false`，`audited_exclusions` 记录其模式化原因；
`relations` 只保存显式关系边；`verification` 在 hydrate 后通过同一路径更新数量与失败项。
示例中的零要求是未实例化占位状态，必须保持 `partial`、`coverage_ratio: 0.0` 和阻断缺口；
写入真实清单时必须替换为非空任务契约引用及其实际覆盖结果，零要求不得标记为 `completed`。
任何字段都不得复制来源载荷、核验原文、报告正文或模型推理。
